cmake_minimum_required(VERSION 3.16)
project(eppg_native_regressions LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(DEFAULT_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PROJECT_ROOT "${DEFAULT_PROJECT_ROOT}" CACHE PATH "Controller implementation to test")
set(HOST_TEST_ROOT "${DEFAULT_PROJECT_ROOT}" CACHE PATH "Host test and stub source root")
set(GTEST_DIR "$ENV{GTEST_DIR}" CACHE PATH "GoogleTest source tree")
if(NOT GTEST_DIR AND EXISTS "${PROJECT_ROOT}/.pio/libdeps/native-test/googletest/googletest/src/gtest-all.cc")
  set(GTEST_DIR "${PROJECT_ROOT}/.pio/libdeps/native-test/googletest")
endif()
if(NOT EXISTS "${GTEST_DIR}/googletest/src/gtest-all.cc")
  include(FetchContent)
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG v1.15.2 GIT_SHALLOW TRUE)
  FetchContent_GetProperties(googletest)
  if(NOT googletest_POPULATED)
    FetchContent_Populate(googletest)
  endif()
  set(GTEST_DIR "${googletest_SOURCE_DIR}")
endif()
add_library(gtest STATIC "${GTEST_DIR}/googletest/src/gtest-all.cc")
target_include_directories(gtest PUBLIC "${GTEST_DIR}/googletest/include" PRIVATE "${GTEST_DIR}/googletest")
add_library(cjson STATIC "${HOST_TEST_ROOT}/test/vendor/cjson/cJSON.c")
target_include_directories(cjson PUBLIC "${HOST_TEST_ROOT}/test/vendor/cjson")
add_library(eppg_test_platform STATIC platform.cpp)
target_include_directories(eppg_test_platform PUBLIC
  "${HOST_TEST_ROOT}/test/native_stubs" "${PROJECT_ROOT}/inc" "${PROJECT_ROOT}/src")
enable_testing()
foreach(suite throttle diagnostics hysteresis simplemonitor multilogger wireless_protocol)
  add_executable(test_${suite} "${HOST_TEST_ROOT}/test/test_${suite}/test_${suite}.cpp")
  target_link_libraries(test_${suite} PRIVATE gtest cjson eppg_test_platform)
  target_compile_definitions(test_${suite} PRIVATE PIO_UNIT_TESTING=1 EPPG_NATIVE_IDF=1)
  add_test(NAME ${suite} COMMAND test_${suite})
endforeach()
